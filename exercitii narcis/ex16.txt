abstract class Forma {
    public abstract double aria();
}

class Cerc extends Forma {
    private double raza;

    public Cerc(double raza) {
        this.raza = raza;
    }

    public double aria() {
        return Math.PI * raza * raza;
    }
}


class Dreptunghi extends Forma {
    private double latime;
    private double lungime;

    public Dreptunghi(double latime, double lungime) {
        this.latime = latime;
        this.lungime = lungime;
    }
    
    public double aria() {
        return latime * lungime;
    }
}

public class Main {
    public static void main(String[] args) {
        Forma[] forme = new Forma[2];
        forme[0] = new Cerc(5);
        forme[1] = new Dreptunghi(4, 7);

        for (Forma f : forme) {
            System.out.println("Aria este: " + f.aria());
        }
    }
}
